cmake_minimum_required(VERSION 3.28)
project(cps2_inputlab_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(SDL3 REQUIRED CONFIG)
find_package(ZLIB REQUIRED)
get_filename_component(CPS2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
foreach(kind keyboard gamepad input_profile)
    add_executable(${kind}_tests "${CPS2_ROOT}/port/${kind}_tests.cpp")
    target_link_libraries(${kind}_tests PRIVATE "${INPUTLAB_CORE_LIBRARY}" SDL3::SDL3 ZLIB::ZLIB cfgmgr32 ws2_32 iphlpapi)
    add_test(NAME ${kind} COMMAND ${kind}_tests)
endforeach()

add_executable(ssf2t_colour_tests ssf2t_colour_tests.cpp "${CPS2_ROOT}/games/ssf2t/src/inputlab_cheats.cpp"
    "${CPS2_ROOT}/runtime/no_blocks.cpp" "${CPS2_ROOT}/runtime/no_z80_blocks.cpp")
target_include_directories(ssf2t_colour_tests PRIVATE "${CPS2_ROOT}/runtime" "${CPS2_ROOT}/machine" "${CPS2_ROOT}/games/ssf2t/src")
target_link_libraries(ssf2t_colour_tests PRIVATE "${INPUTLAB_CORE_LIBRARY}" SDL3::SDL3 ZLIB::ZLIB cfgmgr32 ws2_32 iphlpapi)
add_executable(collection_input_tests collection_input_tests.cpp)
target_link_libraries(collection_input_tests PRIVATE "${INPUTLAB_CORE_LIBRARY}" SDL3::SDL3 ZLIB::ZLIB cfgmgr32 ws2_32 iphlpapi)
add_test(NAME collection_input COMMAND collection_input_tests)
add_executable(pad_hotkey_tests pad_hotkey_tests.cpp)
target_link_libraries(pad_hotkey_tests PRIVATE "${INPUTLAB_CORE_LIBRARY}" SDL3::SDL3 ZLIB::ZLIB cfgmgr32 ws2_32 iphlpapi)
add_test(NAME pad_hotkeys COMMAND pad_hotkey_tests)
add_executable(equalizer_tests equalizer_tests.cpp)
target_link_libraries(equalizer_tests PRIVATE "${INPUTLAB_CORE_LIBRARY}" SDL3::SDL3 ZLIB::ZLIB cfgmgr32 ws2_32 iphlpapi)
add_test(NAME equalizer COMMAND equalizer_tests)
add_executable(sfa2_damage_tests "${CPS2_ROOT}/games/sfa2/tests/damage_tests.cpp" "${CPS2_ROOT}/games/sfa2/src/damage.cpp"
    "${CPS2_ROOT}/runtime/no_blocks.cpp" "${CPS2_ROOT}/runtime/no_z80_blocks.cpp")
target_include_directories(sfa2_damage_tests PRIVATE "${CPS2_ROOT}/runtime" "${CPS2_ROOT}/port" "${CPS2_ROOT}/machine" "${CPS2_ROOT}/fight" "${CPS2_ROOT}/games/sfa2")
target_compile_definitions(sfa2_damage_tests PRIVATE INPUTLAB_SFA2)
target_link_libraries(sfa2_damage_tests PRIVATE "${INPUTLAB_CORE_LIBRARY}" SDL3::SDL3 ZLIB::ZLIB cfgmgr32 ws2_32 iphlpapi)

foreach(game sfa sfz2al ssf2t)
    if(game STREQUAL "ssf2t")
        set(lab_source "${CPS2_ROOT}/games/${game}/src/inputlab_cheats.cpp")
    else()
        set(lab_source "${CPS2_ROOT}/games/${game}/src/inputlab_damage.cpp")
    endif()
    add_executable(${game}_policy_tests damage_port_tests.cpp ${lab_source}
        "${CPS2_ROOT}/runtime/no_blocks.cpp" "${CPS2_ROOT}/runtime/no_z80_blocks.cpp")
    string(TOUPPER "${game}" game_define)
    target_compile_definitions(${game}_policy_tests PRIVATE "LAB_${game_define}")
    target_include_directories(${game}_policy_tests PRIVATE "${CPS2_ROOT}/runtime" "${CPS2_ROOT}/machine"
        "${CPS2_ROOT}/games/${game}" "${CPS2_ROOT}/games/sfa2/src" "${CPS2_ROOT}/fight")
    target_link_libraries(${game}_policy_tests PRIVATE "${INPUTLAB_CORE_LIBRARY}" SDL3::SDL3 ZLIB::ZLIB cfgmgr32 ws2_32 iphlpapi)
endforeach()
